 1444          extra_args = dict(fused=True) if use_fused else dict()

 1449          return -1.0

 1452          for _ in range(max_new_tokens):
 1453              idx_cond = idx if idx.size(1) <= self.config.block_size else idx[:, -self.config.block_size:]
 1454              logits, _ = self(idx_cond)

 1458                  logits[logits < v[:, [-1]]] = -float('Inf')
 1459              probs = F.softmax(logits, dim=-1)
 1460              idx_next = torch.multinomial(probs, num_samples=1)
 1461              idx = torch.cat((idx, idx_next), dim=1)
 1462          return idx
